        w = QWidget()

        w.setStyleSheet(f"background: {C.DARK}; border-right: 1px solid {C.BORDER};")
        lay = QVBoxLayout(w)

        lay.addSpacing(10)
        lay.setContentsMargins(8, 10, 8, 10)

        hdr = QLabel("◈ SYS MONITOR")

            f"border-bottom: 1px solid {C.BORDER}; padding-bottom: 4px;"
        )
        lay.addWidget(hdr)
